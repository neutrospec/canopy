def execute(argv, **kw):
    argv=list(map(str,argv))
    cp=subprocess.run(argv,env=ENV,text=True,capture_output=True,timeout=kw.pop('timeout',25),**kw)
    TRACE.write(json.dumps(dict(argv=argv,exit=cp.returncode,stdout=cp.stdout,stderr=cp.stderr),ensure_ascii=False)+'\n');TRACE.flush()
    return cp
def cli(w,*args,**kw):return execute([BIN,'--wiki',w,*args],**kw)
def ok(w,*args,**kw):
    r=cli(w,*args,**kw);assert r.returncode==0,(args,r.stdout,r.stderr);return r
def j(w,*args):return json.loads(ok(w,*args,'--json').stdout)
def emit(case,**obs):print(json.dumps(dict(case=case,**obs),ensure_ascii=False),flush=True)
def write(p,s):p.parent.mkdir(parents=True,exist_ok=True);p.write_text(s);return p
def page(w,slug='reed',body='INITIAL-UNIQUE',typ='concept',dir='concepts'):
    return write(w/dir/(slug+'.md'),f'---\ntitle: {slug}\ncreated: 2026-02-11\nupdated: 2026-02-11\ntype: {typ}\ntags: [tool]\nsources: []\n---\n\n{body}\n')
def fresh(name,body='INITIAL-UNIQUE'):

def bodyfile(name,text):return write(ROOT/'inputs'/name,text)
def logs(w):return sum(len(p.read_text().splitlines()) for p in (w/'logs').glob('*.jsonl'))
def git(w,*args):
    r=execute(['git','-C',w,*args]);assert r.returncode==0,(args,r.stderr);return r.stdout.strip()
def initgit(w):
    git(w,'init','-b','main');git(w,'config','user.name','Red Data Fixture');git(w,'config','user.email','red-data@example.invalid')
    git(w,'add','-A');git(w,'commit','-m','fixture baseline')
def peers(name):
    w=fresh(name);ok(w,'update','reed','--json');initgit(w)
    origin=ROOT/'fixtures'/(name+'-origin.git');peer=ROOT/'fixtures'/(name+'-peer')
    r=execute(['git','init','--bare','-b','main',origin]);assert not r.returncode
    git(w,'remote','add','origin',origin);git(w,'push','-u','origin','main')
    r=execute(['git','clone',origin,peer]);assert not r.returncode
    git(peer,'config','user.name','Other Fixture');git(peer,'config','user.email','peer@example.invalid')
    return w,origin,peer
def blocked(w,verb,*args):
    fifo=ROOT/'inputs'/(w.name+'-fifo');os.mkfifo(fifo)
    argv=list(map(str,[BIN,'--wiki',w,verb,*args,'--body-file',fifo,'--json']))
    p=subprocess.Popen(argv,env=ENV,text=True,stdout=subprocess.PIPE,stderr=subprocess.PIPE)
    deadline=time.monotonic()+10
    while time.monotonic()<deadline:
        try:return p,os.open(fifo,os.O_WRONLY|os.O_NONBLOCK),argv
        except OSError as e:
            assert e.errno==errno.ENXIO and p.poll() is None
            time.sleep(.005)
    p.kill();p.communicate();raise AssertionError('reader never reached body input')
def release(p,fd,argv,text):
    os.write(fd,text.encode());os.close(fd);out,err=p.communicate(timeout=25)
    TRACE.write(json.dumps(dict(argv=argv,exit=p.returncode,stdout=out,stderr=err))+'\n');TRACE.flush()
    return p.returncode
