57				flush()
58				runes := []rune(p)

60					end := min(start+targetRunes, len(runes))
61					cur.WriteString(string(runes[start:end]))
62					curLen = end - start
63					flush()
64				}
65				continue
66			}
67			if curLen > 0 && curLen+n > targetRunes {
68				flush()
69			}
70			if curLen > 0 {
71				cur.WriteString("\n\n")
72			}
73			cur.WriteString(p)

75		}
