records = []
def run(label, argv, changes=None, cwd=None, timeout=60):
    e = dict(env)
    if changes: e.update(changes)
    start = time.monotonic()
    try:
        p = subprocess.run(argv, cwd=cwd or REPO, env=e, capture_output=True, text=True, timeout=timeout)
        row = dict(label=label, argv=argv, cwd=str(cwd or REPO), exit=p.returncode,
                   elapsed_ms=int((time.monotonic()-start)*1000), stdout=p.stdout, stderr=p.stderr)
    except subprocess.TimeoutExpired as x:
        row = dict(label=label, argv=argv, exit='timeout', stdout=str(x.stdout), stderr=str(x.stderr))
    records.append(row)
    print(json.dumps(row, ensure_ascii=False), flush=True)
    (BASE/'logs'/'checks.json').write_text(json.dumps(records, ensure_ascii=False, indent=2))
    return row
