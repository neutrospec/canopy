from pathlib import Path
import hashlib
import json
import os
import shlex
import subprocess
import tempfile

BASE = Path('/private/tmp/canopy-audit-20261009/docs')
RUN = Path(tempfile.mkdtemp(prefix='extra-', dir=BASE))
BIN = BASE/'canopy'
ENV = os.environ.copy()
ENV.update(HOME=str(RUN/'home'), XDG_CONFIG_HOME=str(RUN/'config'),
           XDG_CACHE_HOME=str(RUN/'cache'), XDG_DATA_HOME=str(RUN/'data'),
           XDG_STATE_HOME=str(RUN/'state'), GOCACHE='/private/tmp/canopy-review/go-cache',
           GOPROXY='off', GOTOOLCHAIN='local', GIT_CONFIG_NOSYSTEM='1',
           GIT_CONFIG_GLOBAL='/dev/null')
ENV.pop('CANOPY_WIKI', None)
ENV.pop('PI_CODING_AGENT_DIR', None)
(RUN/'home').mkdir()
RECORDS = []

def command(args, cwd=RUN, stdin=None):
    p=subprocess.run(args,cwd=cwd,env=ENV,input=stdin,text=True,capture_output=True,timeout=45)
    r=dict(argv=[str(a) for a in args],cwd=str(cwd),exit=p.returncode,stdout=p.stdout,stderr=p.stderr)
    RECORDS.append(r)
    return r
def canopy(*args,cwd=RUN,stdin=None):
    return command([str(BIN),*args],cwd=cwd,stdin=stdin)
def h(p):
    return hashlib.sha256(p.read_bytes()).hexdigest() if p.exists() else None

w=RUN/'wiki-a';w.mkdir()
canopy('init','--wiki',str(w),'--json')
canopy('tags','--audit','--wiki',str(w),'--json')
w2=RUN/'wiki-b';w2.mkdir()
canopy('init','--wiki',str(w2),'--json')
command(['git','init','-b','main',str(w2)])
outside=canopy('status','--json')
inside=canopy('status','--json',cwd=w2)
RECORDS.append(dict(probe='wiki_resolution',default_root=json.loads(outside['stdout'])['root'],
                    from_other_wiki_root=json.loads(inside['stdout'])['root']))

# No sync/commit invocation in this suite. All changes are disposable fixture files.
canopy('new','Alpha','--slug','alpha','--type','concept','--tags','tool','--body-file','-',
       '--wiki',str(w),'--json',stdin='Alpha body.\n')
logs=sorted((w/'logs').glob('*.jsonl')); log=logs[-1]
lines=log.read_text().splitlines()
entry=json.loads(lines[0]); entry['note']='audit-edited-history'
lines[0]=json.dumps(entry,ensure_ascii=False)
log.write_text('\n'.join(lines)+'\n')
canopy('update','alpha','--wiki',str(w),'--json')
RECORDS.append(dict(probe='logs_not_regenerated',first_note=json.loads(log.read_text().splitlines()[0])['note'],
                    generated_marker_present='<!-- generated by canopy' in log.read_text()))

canopy('show','alpha','--wiki',str(w))
canopy('new','Beta','--slug','beta','--type','concept','--wiki',str(w),'--json')
co=canopy('checkout','beta','--wiki',str(w),'--json')
canopy('list','--wiki',str(w),'--json')
canopy('show','alpha','--wiki',str(w),'--no-mark')
canopy('checkin','beta','--wiki',str(w),'--discard','--json')

literal='audit_w='+shlex.quote('--wiki '+str(w))+'\n'+shlex.quote(str(BIN))+' lint $audit_w --json'
command(['zsh','-c',literal])
literal2='audit_w='+shlex.quote('--wiki '+str(w2))+'\ngit -C $audit_w status --short'
command(['bash','-c',literal2])

# Run the skill's unconditional new -> show -> rm sequence with a parser
# rejection. This is a skill-example failure probe, not an agent simulation.
body=RUN/'example-body.md'
body.write_text('```mermaid\nflowchart LR\nA[missing --> B\n```\n')
script='\n'.join([
 'body_file='+shlex.quote(str(body)),
 shlex.quote(str(BIN))+' new '+shlex.quote('새 개념')+' --type concept --slug new-concept --body-file "$body_file" --wiki '+shlex.quote(str(w)),
 shlex.quote(str(BIN))+' show new-concept --wiki '+shlex.quote(str(w)),
 'rm "$body_file"'])
example=command(['zsh','-c',script])
RECORDS.append(dict(probe='skill_example_failure',shell_exit=example['exit'],body_exists=body.exists(),
                    page_exists=(w/'concepts/new-concept.md').exists()))

# Existing corpus changes after the mutation has landed but before its derived
# effects finish. A retrying agent must inspect the page instead of assuming
# exit != 0 means 'nothing happened'.
w3=RUN/'wiki-partial';w3.mkdir()
canopy('init','--wiki',str(w3),'--json')
(w3/'logs').write_text('fixture obstruction')
failure=canopy('new','Partial','--slug','partial','--type','concept','--body-file','-',
               '--wiki',str(w3),'--json',stdin='Only copy of the intended content.\n')
RECORDS.append(dict(probe='partial_failure',exit=failure['exit'],page_exists=(w3/'concepts/partial.md').exists(),
                    index_exists=(w3/'index.md').exists()))
canopy('new','Partial','--slug','partial','--type','concept','--wiki',str(w3),'--json')

# New-format init has topics/forms; the legacy FAQ's advice to add a tags list
# creates a taxonomy branch that is accepted but absent from facet governance.
cfg=w/'canopy.toml'
txt=cfg.read_text().replace('[schema]','[schema]\n  tags = ["audit-topic"]',1)
cfg.write_text(txt)
canopy('tags','--wiki',str(w),'--json')
canopy('tags','--audit','--wiki',str(w),'--json')

(BASE/'extra-results.json').write_text(json.dumps(dict(run=str(RUN),records=RECORDS),ensure_ascii=False,indent=2))
for r in RECORDS:
    if 'argv' in r:
        r=dict(command=r['argv'][1:] if r['argv'][0]==str(BIN) else r['argv'],exit=r['exit'],stdout=r['stdout'][:1300],stderr=r['stderr'][:900])
    print(json.dumps(r,ensure_ascii=False))
