var fixtureRoot string
var cli string
var fixedNow = time.Date(2026, 10, 9, 12, 0, 0, 0, time.UTC)

func must(err error) {
	if err != nil {
		panic(err)
	}
}

func write(path, body string) {
	must(os.MkdirAll(filepath.Dir(path), 0o755))
	must(os.WriteFile(path, []byte(body), 0o644))
}

func read(path string) string {
	b, err := os.ReadFile(path)
	must(err)
	return string(b)
}

func fresh(label string) *config.Wiki {
	root, err := os.MkdirTemp(fixtureRoot, label+"-")
	must(err)
	return &config.Wiki{Root: root, Cfg: config.Default()}
}

func page(w *config.Wiki, slug, body string) string {
	rel := "concepts/" + slug + ".md"
	write(filepath.Join(w.Root, rel), fmt.Sprintf("---\ntitle: %s\ncreated: 2026-01-01\nupdated: 2026-01-01\ntype: concept\ntags: [tool]\n---\n\n%s\n", slug, body))
	return rel
}

func scan(w *config.Wiki) *wiki.ScanResult {
	s, err := wiki.Scan(w)
	must(err)
	return s
}

func state(w *config.Wiki) *reads.State {
	s, err := reads.Load(w)
	must(err)
	return s
}

func rs(w *config.Wiki) *resurface.State {
	s, err := resurface.LoadState(w)
	must(err)
	return s
}

func jsonField(path, field string) string {
	var m map[string]json.RawMessage
	must(json.Unmarshal([]byte(read(path)), &m))
	if len(m[field]) == 0 {
		return "<missing>"
	}
	return string(m[field])
}

func command(w *config.Wiki, args ...string) (string, int) {
	cmd := exec.Command(cli, append(args, "--wiki", w.Root)...)
	b, err := cmd.CombinedOutput()
	if err == nil {
		return string(b), 0
	}
	if e, ok := err.(*exec.ExitError); ok {
		return string(b), e.ExitCode()
	}
	panic(err)
}

func atomicChild(root string) {
	w := &config.Wiki{Root: root, Cfg: config.Default()}
	s := state(w)
	s.Mark("beta", "explicit", fixedNow)
	var lim syscall.Rlimit
	must(syscall.Getrlimit(syscall.RLIMIT_FSIZE, &lim))
	old := lim
	lim.Cur = 128
	signal.Ignore(syscall.SIGXFSZ)
	must(syscall.Setrlimit(syscall.RLIMIT_FSIZE, &lim))
	err := s.Save(w)
	must(syscall.Setrlimit(syscall.RLIMIT_FSIZE, &old))
	_, loadErr := reads.Load(w)
	st, statErr := os.Stat(reads.Path(w))
	must(statErr)
	fmt.Printf("faulted Save=%v bytes=%d subsequent Load=%v\n", err, st.Size(), loadErr)
}

func concurrentEdits() {
	w := fresh("web-edit-concurrent")
	page(w, "topic", "original")
	server, err := webui.NewServer(w, nil)
	must(err)
	handler := server.Handler()
	const n = 48
	start := make(chan struct{})
	status := make(chan int, n)
	var wg sync.WaitGroup
	for i := 0; i < n; i++ {
		wg.Add(1)
		go func(i int) {
			defer wg.Done()
			<-start
			values := url.Values{"body": {fmt.Sprintf("unique-proposed-body-%03d", i)}}
			r := httptest.NewRequest("POST", "/edit/topic", strings.NewReader(values.Encode()))
			r.Header.Set("Content-Type", "application/x-www-form-urlencoded")
			rw := httptest.NewRecorder()
			handler.ServeHTTP(rw, r)
			status <- rw.Code
		}(i)
	}
	close(start)
	wg.Wait()
	close(status)
	codes := map[int]int{}
	for code := range status {
		codes[code]++
	}
	entries, err := os.ReadDir(tasks.Dir(w))
	must(err)
	valid, bodies, corrupt := 0, map[string]bool{}, 0
	for _, e := range entries {
		var task tasks.Task
		if json.Unmarshal([]byte(read(filepath.Join(tasks.Dir(w), e.Name()))), &task) != nil {
			corrupt++
			continue
		}
		valid++
		bodies[task.Body] = true
	}
	fmt.Printf("web_requests=%d HTTP=%v task_files=%d valid_files=%d distinct_preserved_bodies=%d corrupt=%d\n", n, codes, len(entries), valid, len(bodies), corrupt)
	left, right := fresh("device-a"), fresh("device-b")
	l, err := tasks.FileEdit(left, "topic", "", "proposal A", "base", "web", fixedNow)
	must(err)
	r, err := tasks.FileEdit(right, "topic", "", "proposal B", "base", "web", fixedNow)
	must(err)
	fmt.Printf("independent_device_ids_equal=%v id=%s\n", l.ID == r.ID, l.ID)
}

func panicOf(fn func()) (p any) {
	defer func() { p = recover() }()
	fn()
	return nil
}

	write(metaPath, "{")
	open, err := checkout.Open(w)
	must(err)
	info, err := checkout.Checkout(w, scan(w), "alpha")

	fmt.Printf("corrupt-meta open_count=%d Checkout_error=nil Modified=%v edited_copy_preserved=%v wiki_unchanged=%v\n", len(open), info.Modified, strings.Contains(read(i.Path), "uncommitted-only-copy"), read(filepath.Join(w.Root, rel)) == before)
	// A missing meta has the same overwrite route.
	write(i.Path, strings.Replace(before, "original", "second-only-copy", 1))
	must(os.Remove(metaPath))
	_, err = checkout.Checkout(w, scan(w), "alpha")
	must(err)
	fmt.Printf("missing-meta edited_copy_preserved=%v\n", strings.Contains(read(i.Path), "second-only-copy"))
	must(os.Chmod(w.CheckoutDir(), 0o555))
	discardErr := checkout.Discard(w, "alpha")
	_, copyErr := os.Stat(i.Path)
	_, metaErr := os.Stat(metaPath)
	must(os.Chmod(w.CheckoutDir(), 0o755))
	fmt.Printf("unwritable_checkout_dir Discard_error=%v copy_still_exists=%v meta_still_exists=%v\n", discardErr, copyErr == nil, metaErr == nil)
}

func mixedOffsets() {

	must(err)
	defer e.Close()
	kst := time.FixedZone("KST", 9*3600)
	older := time.Date(2026, 10, 9, 9, 0, 0, 0, kst) // 00:00Z
	newer := time.Date(2026, 10, 9, 5, 0, 0, 0, time.UTC)
	must(e.Log(older, "older", "agent", "show", ""))
	must(e.Log(newer, "newer", "agent", "show", ""))
	q, err := e.Query(attention.Filter{Since: time.Date(2026, 10, 9, 4, 0, 0, 0, time.UTC)})
	must(err)
	var slugs []string
	for _, event := range q {
		slugs = append(slugs, event.Slug)
	}
	fmt.Printf("since=2026-10-09T04:00:00Z expected=[newer] actual=%v\n", slugs)
	n, err := e.Prune(time.Date(2026, 10, 9, 10, 0, 0, 0, kst)) // 01:00Z
	must(err)
	q, err = e.Recent(10)
	must(err)
	slugs = nil
	for _, event := range q {
		slugs = append(slugs, event.Slug)
	}
	fmt.Printf("prune_before=2026-10-09T10:00:00+09:00 removed=%d expected_remaining=[newer] actual=%v\n", n, slugs)
	s := state(w)
	s.Mark("older", "explicit", older)
	s.MarkAgent("newer", newer)
	fmt.Printf("RecentAccessSlugs expected=[newer older] actual=%v\n", s.RecentAccessSlugs(2))
}

func migrationsAndDB() {
	for _, input := range []string{"missing", `{"schema_version":99}`, "{", "null", `{"schema_version":-1}`} {
		w := fresh("migrate")
		ctx := &migrate.Context{ConfigHome: filepath.Join(w.Root, "config"), CacheHome: filepath.Join(w.Root, "cache"), DataHome: filepath.Join(w.Root, "data")}
		if input != "missing" {
			write(filepath.Join(ctx.ConfigHome, "state.json"), input)
		}
		result, err := migrate.Ensure(ctx, "audit")
		second, secondErr := migrate.Ensure(ctx, "audit")
		fmt.Printf("migrate input=%s result=%+v error=%v second=%+v second_error=%v\n", input, result, err, second, secondErr)
	}
	w := fresh("migrate-concurrent")
	ctx := &migrate.Context{ConfigHome: filepath.Join(w.Root, "config"), CacheHome: filepath.Join(w.Root, "cache"), DataHome: filepath.Join(w.Root, "data")}
	const n = 32
	errs := make(chan error, n)
	start := make(chan struct{})
	for i := 0; i < n; i++ {
		go func() { <-start; _, err := migrate.Ensure(ctx, "audit"); errs <- err }()
	}
	close(start)
	fails := 0
	var example error
	for i := 0; i < n; i++ {
		if err := <-errs; err != nil {
			fails++
			example = err
		}
	}
	fmt.Printf("fresh concurrent Ensure calls=%d failed=%d example=%v\n", n, fails, example)
	// Open touches a database before looking at its local schema version.
	futurePath := filepath.Join(w.Root, "future.db")
	db, err := sql.Open("sqlite", futurePath)
	must(err)
	_, err = db.Exec("PRAGMA user_version = 99")
	must(err)
	must(db.Close())
	ev, err := attention.Open(futurePath)
	fmt.Printf("future attention DB user_version=99 Open_error=%v\n", err)
	if err == nil {
		fmt.Printf("future attention DB Log_error=%v\n", ev.Log(fixedNow, "alpha", "agent", "show", ""))
		must(ev.Close())
	}
	db, err = sql.Open("sqlite", futurePath)
	must(err)
	defer db.Close()
	var v, tables int
	must(db.QueryRow("PRAGMA user_version").Scan(&v))
	must(db.QueryRow("SELECT COUNT(*) FROM sqlite_master WHERE type='table' AND name='events'").Scan(&tables))
	fmt.Printf("future attention DB after Open version=%d events_table_created=%v\n", v, tables == 1)
}

func migrationInterruption() {

}
