BIN=ROOT/'canopy-fixed'
def sha(path):return hashlib.sha256(path.read_bytes()).hexdigest() if path.is_file() else None
def page(name,body):return f'---\ntitle: "{name}"\ntype: concept\ncreated: 2026-01-01\nupdated: 2026-01-01\ntags: []\n---\n\n{body}\n'
def seed(name,names):
 w=ROOT/'fixtures'/name;(w/'concepts').mkdir(parents=True)
 for n in names:(w/'concepts'/f'{n}.md').write_text(page(n,'ORIGINAL '+n))
 r=run(name+'-init',[BIN,'--wiki',w,'init','--json']);assert r['exit']==0,r
 return w

out={'checkout_recovery':recovery,'orphan':orphan_result,'queue':queue,'rename':rename};(ROOT/'recovery-queue-probes.json').write_text(json.dumps(out,indent=2)+'\n');print(json.dumps({k:v['passes'] for k,v in out.items()},indent=2));print('GC',queue['gc']['passes']);print('rename control',rename['control']['passes'])
