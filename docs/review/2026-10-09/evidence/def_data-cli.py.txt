def hold(w,verb,*a):
    fifo=ROOT/'inputs'/(w.name+'-fifo'); os.mkfifo(fifo)
    argv=[BIN,'--wiki',str(w),verb,*a,'--body-file',str(fifo),'--json']
    p=subprocess.Popen(argv,stdout=subprocess.PIPE,stderr=subprocess.PIPE,text=True)
    end=time.monotonic()+8
    while time.monotonic()<end:
        try: return p,os.open(fifo,os.O_WRONLY|os.O_NONBLOCK),argv
        except OSError:
            if p.poll() is not None: raise RuntimeError(p.communicate())
            time.sleep(.01)
    p.kill(); p.communicate(); raise RuntimeError('FIFO reader did not arrive')
def release(p,fd,argv,body):
    os.write(fd,body.encode()); os.close(fd); out,err=p.communicate(timeout=15)
    record(argv,subprocess.CompletedProcess(argv,p.returncode,out,err)); return p.returncode
