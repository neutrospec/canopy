type response struct{Code int;Body string;Header http.Header;FinalPath string}
func call(t *testing.T,c *http.Client,method,target string,body io.Reader,headers map[string]string) response{
 t.Helper();req,e:=http.NewRequest(method,target,body);if e!=nil{t.Fatal(e)}
 for k,v:=range headers{if k=="Host"{req.Host=v}else{req.Header.Set(k,v)}}
 res,e:=c.Do(req);if e!=nil{t.Fatal(e)};defer res.Body.Close();b,e:=io.ReadAll(res.Body);if e!=nil{t.Fatal(e)};return response{res.StatusCode,string(b),res.Header,res.Request.URL.Path}
}
func nofollow()*http.Client{return &http.Client{Timeout:30*time.Second,CheckRedirect:func(*http.Request,[]*http.Request)error{return http.ErrUseLastResponse}}}
