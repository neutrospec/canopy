def run(argv, cwd=REPO, stdin=None, extra=None, timeout=120):
    env=ENV.copy(); env.update(extra or {})
    before=time.monotonic()
    p=subprocess.run([str(a) for a in argv],cwd=cwd,env=env,input=stdin,
                     capture_output=True,text=True,timeout=timeout)
    rec=dict(argv=[str(a) for a in argv],cwd=str(cwd),exit=p.returncode,
             seconds=round(time.monotonic()-before,3),stdout=p.stdout,stderr=p.stderr)
    RECORDS.append(rec)
    return rec

def obs(name,**data):
