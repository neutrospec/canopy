func fixture(t *testing.T, body string) (*webui.Server, *config.Wiki) {
	t.Helper()
	root := t.TempDir()
	os.MkdirAll(filepath.Join(root, "concepts"), 0755)
	page := "---\ntitle: Alpha\ntype: concept\ncreated: 2020-01-01\nupdated: 2020-01-01\ntags: [tool]\n---\n" + body
	if err := os.WriteFile(filepath.Join(root, "concepts", "alpha.md"), []byte(page), 0644); err != nil {
		t.Fatal(err)
	}
	w := &config.Wiki{Root: root, Cfg: config.Default()}
	s, err := webui.NewServer(w, nil)
	if err != nil {
		t.Fatal(err)
	}
	return s, w
}
func req(h http.Handler, method, path, host, origin, body string) *httptest.ResponseRecorder {
	r := httptest.NewRequest(method, path, strings.NewReader(body))
	if host != "" {
		r.Host = host
	}
	if origin != "" {
		r.Header.Set("Origin", origin)
	}
	r.Header.Set("Content-Type", "application/x-www-form-urlencoded")
	rec := httptest.NewRecorder()
	h.ServeHTTP(rec, r)
	return rec
}

func TestWikiScale(t *testing.T) {
	for _, n := range []int{100, 1000, 5000} {
		s, w := fixture(t, "unique-queryword original\n")
		for i := 1; i < n; i++ {
			slug := fmt.Sprintf("page-%05d", i)
			body := "---\ntitle: " + slug + "\ntype: concept\ncreated: 2020-01-01\nupdated: 2020-01-01\ntags: [tool]\n---\n" + strings.Repeat("fixture-text ", 320) + "\n[[alpha]]\n"
			if err := os.WriteFile(filepath.Join(w.Root, "concepts", slug+".md"), []byte(body), 0644); err != nil {
				t.Fatal(err)
			}
		}
		h := s.Handler()
		for _, p := range []string{"/api/preview/alpha", "/api/search?q=fixture", "/browse", "/api/graph", "/"} {
			var ds []time.Duration
			var allocs []uint64
			out := 0
			for j := 0; j < 3; j++ {
				var a, b runtime.MemStats
				runtime.ReadMemStats(&a)
				start := time.Now()
				r := req(h, "GET", p, "localhost:8737", "", "")
				ds = append(ds, time.Since(start))
				runtime.ReadMemStats(&b)
				allocs = append(allocs, b.TotalAlloc-a.TotalAlloc)
				out = r.Body.Len()
				if r.Code != 200 {
					t.Fatal(r.Code, r.Body.String())
				}
			}
			sort.Slice(ds, func(i, j int) bool { return ds[i] < ds[j] })
			sort.Slice(allocs, func(i, j int) bool { return allocs[i] < allocs[j] })
			t.Logf("pages=%d path=%s median=%v allocatedBytes=%d responseBytes=%d", n, p, ds[1], allocs[1], out)
		}
	}
}

func TestUnauthMultipartSpool(t *testing.T) {
	s, _ := fixture(t, "fixture\n")
	t.Setenv("XDG_CONFIG_HOME", t.TempDir())
	tmp := t.TempDir()
	t.Setenv("TMPDIR", tmp)
	if _, err := s.EnableAuth(); err != nil {
		t.Fatal(err)
	}
	boundary := "audit-boundary"
	prefix := "--" + boundary + "\r\nContent-Disposition: form-data; name=\"upload\"; filename=\"fixture.bin\"\r\nContent-Type: application/octet-stream\r\n\r\n"
	suffix := "\r\n--" + boundary + "--\r\n"
	r := httptest.NewRequest("POST", "/setup", io.MultiReader(strings.NewReader(prefix), strings.NewReader(strings.Repeat("z", 64<<20)), strings.NewReader(suffix)))
	r.Header.Set("Content-Type", "multipart/form-data; boundary="+boundary)
	rec := httptest.NewRecorder()
	s.Handler().ServeHTTP(rec, r)
	var total int64
	entries, _ := os.ReadDir(tmp)
	for _, e := range entries {
		if st, err := e.Info(); err == nil {
			total += st.Size()
		}
	}
	t.Logf("unauth setup suppliedFileBytes=%d status=%d tempfileCount=%d spooledBytes=%d", 64<<20, rec.Code, len(entries), total)
	if r.MultipartForm != nil {
		r.MultipartForm.RemoveAll()
	}
}
