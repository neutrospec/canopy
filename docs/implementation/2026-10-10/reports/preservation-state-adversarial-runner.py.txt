from pathlib import Path
import os, subprocess, json, sys, time
ROOT=Path(__file__).resolve().parent
REPO=Path('/Users/nobocop/play/canopy')
ENV={k:v for k,v in os.environ.items() if k in ['PATH','LANG','LC_ALL','TZ','TERM']}
ENV.update(HOME=str(ROOT/'home'),XDG_CONFIG_HOME=str(ROOT/'config'),XDG_CACHE_HOME=str(ROOT/'cache'),XDG_DATA_HOME=str(ROOT/'data'),XDG_STATE_HOME=str(ROOT/'state'),TMPDIR=str(ROOT/'tmp'),GOCACHE=str(ROOT/'go-cache'),GOMODCACHE='/Users/nobocop/go/pkg/mod',GOTELEMETRY='off',GOPROXY='off',GOSUMDB='off',GOTOOLCHAIN='local',GOFLAGS='-mod=readonly',CGO_ENABLED='0',GIT_CONFIG_NOSYSTEM='1',GIT_CONFIG_GLOBAL='/dev/null',GIT_AUTHOR_NAME='review-fixture',GIT_AUTHOR_EMAIL='fixture@example.invalid',GIT_COMMITTER_NAME='review-fixture',GIT_COMMITTER_EMAIL='fixture@example.invalid',CANOPY_WIKI='',PI_CODING_AGENT_DIR='')
PREFIX=['/usr/bin/sandbox-exec','-f',str(ROOT/'sandbox.sb')]
def run(label,args,cwd=REPO,timeout=120,input=None):
 start=time.time()
 p=subprocess.run(PREFIX+list(map(str,args)),cwd=cwd,env=ENV,input=input,stdout=subprocess.PIPE,stderr=subprocess.PIPE,text=True,timeout=timeout)
 record={'label':label,'argv':list(map(str,args)),'cwd':str(cwd),'exit':p.returncode,'stdout':p.stdout,'stderr':p.stderr,'seconds':round(time.time()-start,4)}
 (ROOT/'logs'/(label+'.json')).write_text(json.dumps(record,indent=2)+'\n')
 return record
if __name__=='__main__':
 r=run(sys.argv[1],sys.argv[2:],timeout=240)
 print(json.dumps(r,indent=2))
 sys.exit(r['exit'])
