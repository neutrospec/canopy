def command(args, cwd=RUN, stdin=None, extra=None, timeout=90):
    env = ENV.copy()
    if extra:
        env.update(extra)
    p = subprocess.run([str(a) for a in args], cwd=cwd, env=env, input=stdin, capture_output=True, text=True, timeout=timeout)
    r = dict(argv=[str(a) for a in args], cwd=str(cwd), exit=p.returncode, stdout=p.stdout, stderr=p.stderr)
    RECORDS.append(r)
    return r

def obs(name, **data):
    r = dict(probe=name, **data)
