func need(t *testing.T, e error) {
	t.Helper()
	if e != nil {
		t.Fatal(e)
	}
}
func put(t *testing.T, p, s string) {
	t.Helper()
	need(t, os.MkdirAll(filepath.Dir(p), 0755))
	need(t, os.WriteFile(p, []byte(s), 0644))
}
func get(t *testing.T, p string) []byte { t.Helper(); b, e := os.ReadFile(p); need(t, e); return b }
func fresh(t *testing.T) *config.Wiki {
	t.Helper()

}

func TestQ01SnapshotsAndCrossFileControl(t *testing.T) {
	w := fresh(t)
	a, b := rd(t, w), rd(t, w)
	a.Mark("cedar", "explicit", instant)
	b.Mark("birch", "explicit", instant)
	need(t, a.Save(w))
	need(t, b.Save(w))
	t.Logf("two stale human snapshots: cedar=%v birch=%v", rd(t, w).IsRead("cedar"), rd(t, w).IsRead("birch"))
	a, b = rd(t, w), rd(t, w)
	a.MarkAgent("cedar", instant)
	b.MarkAgent("birch", instant)
	need(t, a.Save(w))
	need(t, b.Save(w))
	t.Logf("two stale agent snapshots: cedar=%v birch=%v", rd(t, w).Agent["cedar"] != nil, rd(t, w).Agent["birch"] != nil)
	// Save already has door-specific dirty flags: independent doors do not
	// overwrite each other's file even when both loaded before either save.
	w = fresh(t)
	a, b = rd(t, w), rd(t, w)
	a.Mark("cedar", "explicit", instant)
	b.MarkAgent("birch", instant)
	need(t, a.Save(w))
	need(t, b.Save(w))
	t.Logf("cross-door control: human=%v agent=%v", rd(t, w).IsRead("cedar"), rd(t, w).Agent["birch"] != nil)
	sa, sb := rs(t, w), rs(t, w)
	sa.Snooze("cedar", 5, instant)
	sb.DismissPair("birch", "cedar")
	need(t, sa.Save(w))
	need(t, sb.Save(w))
	t.Logf("stale resurface: snooze=%q dismiss=%q", rs(t, w).Snoozed["cedar"], rs(t, w).ShownPairs["birch|cedar"])
	page(t, w, "cedar", "one")
	page(t, w, "birch", "two")
	_, e := reconcile.BlessAll(w)
	need(t, e)
	la, _, e := reconcile.Load(w)
	need(t, e)
	lb, _, e := reconcile.Load(w)
	need(t, e)
	la.Blessed["concepts/cedar.md"] = "first-new-hash"
	lb.Blessed["concepts/birch.md"] = "second-new-hash"
	need(t, la.Save(w))
	need(t, lb.Save(w))
	lf, _, e := reconcile.Load(w)
	need(t, e)
	t.Logf("stale ledger: first_retained=%v second_retained=%v", lf.Blessed["concepts/cedar.md"] == "first-new-hash", lf.Blessed["concepts/birch.md"] == "second-new-hash")
}

func TestQ03ReachableHTTPConcurrentProposals(t *testing.T) {
	log.SetOutput(io.Discard)
	w := fresh(t)
	page(t, w, "cedar", "old body")
	s, e := webui.NewServer(w, nil)
	need(t, e)
	h := s.Handler()
	submit := func(body string) int {
		r := httptest.NewRequest("POST", "/edit/cedar", strings.NewReader(url.Values{"body": {body}}.Encode()))
		r.Header.Set("Content-Type", "application/x-www-form-urlencoded")
		rw := httptest.NewRecorder()
		h.ServeHTTP(rw, r)
		return rw.Code
	}
	for i := 0; i < 8; i++ {
		if submit(fmt.Sprintf("sequential-%d", i)) != 303 {
			t.Fatal("bad control HTTP")
		}
	}
	first, e := tasks.Load(w)
	need(t, e)
	t.Logf("sequential control: submitted=8 files=%d", len(first))
	const n = 80
	start := make(chan struct{})
	var wg sync.WaitGroup
	codes := make(chan int, n)
	for i := 0; i < n; i++ {
		wg.Add(1)
		go func(i int) { defer wg.Done(); <-start; codes <- submit(fmt.Sprintf("own-distinct-body-%03d", i)) }(i)
	}
	close(start)
	wg.Wait()
	close(codes)
	successes := 0
	for code := range codes {
		if code == 303 {
			successes++
		}
	}
	files, e := os.ReadDir(tasks.Dir(w))
	need(t, e)
	bodies := map[string]bool{}
	corrupt := 0
	for _, f := range files {
		var task tasks.Task
		if json.Unmarshal(get(t, filepath.Join(tasks.Dir(w), f.Name())), &task) != nil {
			corrupt++
			continue
		}
		if strings.HasPrefix(task.Body, "own-distinct-body-") {
			bodies[task.Body] = true
		}
	}
	t.Logf("concurrent actual handler: submitted=%d HTTP303=%d retained_unique_concurrent_bodies=%d files_total=%d corrupt=%d", n, successes, len(bodies), len(files), corrupt)
	l, r := fresh(t), fresh(t)
	lt, e := tasks.FileEdit(l, "cedar", "", "device-one", "base", "web", instant)
	need(t, e)
	rt, e := tasks.FileEdit(r, "cedar", "", "device-two", "base", "web", instant)
	need(t, e)
	t.Logf("independent fixture same second: id_equal=%v", lt.ID == rt.ID)
}

func TestQ04CheckinTimingAndSequentialControl(t *testing.T) {
	w := fresh(t)
	page(t, w, "cedar", "old text")
	sc := scan(t, w)
	info, e := checkout.Checkout(w, sc, "cedar")
	need(t, e)
	cp := string(get(t, info.Path)) + strings.Repeat("\n```mermaid\nflowchart LR\n    A[\"first\"] --> B[\"second\"]\n```\n", 12)
	put(t, info.Path, cp)
	validator := mermaid.NewValidator()
	done := make(chan error, 1)
	start := time.Now()
	go func() { _, e := checkout.Checkin(w, sc, "cedar", validator); done <- e }()
	time.Sleep(25 * time.Millisecond)
	wikiPath := filepath.Join(w.Root, "concepts/cedar.md")
	other := string(get(t, wikiPath)) + "\nANOTHER-WRITER-UNIQUE\n"
	put(t, wikiPath, other)
	e = <-done
	t.Logf("hash->validation->write window: error=%v elapsed=%v other_writer_retained=%v", e, time.Since(start), strings.Contains(string(get(t, wikiPath)), "ANOTHER-WRITER-UNIQUE"))
	w = fresh(t)
	page(t, w, "cedar", "baseline")
	sc = scan(t, w)
	info, e = checkout.Checkout(w, sc, "cedar")
	need(t, e)
	put(t, info.Path, string(get(t, info.Path))+"EDIT")
	page(t, w, "cedar", "earlier other writer")
	_, e = checkout.Checkin(w, sc, "cedar", nil)
	t.Logf("sequential prior base change: rejected=%v copy_present=%v", e != nil, exists(info.Path))
}

func TestQ05MigrationBoundaries(t *testing.T) {
	ctx := func() *migrate.Context {
		w := fresh(t)
		return &migrate.Context{ConfigHome: filepath.Join(w.Root, "config"), CacheHome: filepath.Join(w.Root, "cache"), DataHome: filepath.Join(w.Root, "data")}
	}
	c := ctx()
	const n = 40
	start := make(chan struct{})
	errs := make(chan error, n)
	var wg sync.WaitGroup
	for i := 0; i < n; i++ {
		wg.Add(1)
		go func() { defer wg.Done(); <-start; _, e := migrate.Ensure(c, "review"); errs <- e }()
	}
	close(start)
	wg.Wait()
	close(errs)
	fail := 0
	var example error
	for e := range errs {
		if e != nil {
			fail++
			example = e
		}
	}
	status, se := migrate.Status(c)
	t.Logf("fresh concurrent Ensure=%d failures=%d example=%v final_Status=%+v error=%v", n, fail, example, status, se)
	c = ctx()
	for i := 0; i < 12; i++ {
		_, e := migrate.Ensure(c, "review")
		need(t, e)
	}
	t.Log("sequential Ensure control=12 successful")
	for _, input := range []string{"null", `{"schema_version":-3}`, `{"schema_version":88}`, "{"} {
		c = ctx()
		put(t, filepath.Join(c.ConfigHome, "state.json"), input)
		r, e := migrate.Ensure(c, "review")
		t.Logf("state shape=%s result=%+v error=%v original_preserved=%v", input, r, e, string(get(t, filepath.Join(c.ConfigHome, "state.json"))) == input)
	}
}

	b, e := json.Marshal(raw)
	need(t, e)
	put(t, path, string(b))
	_, e = tasks.Close(w, scan(t, w), task.ID, tasks.StatusDone, "", "cli", instant)

func TestQ12HTTPStateMutationsAndMutexControl(t *testing.T) {
	log.SetOutput(io.Discard)
	for _, door := range []string{"read", "feedback"} {
		w := fresh(t)
		const n = 36
		for i := 0; i < n; i++ {
			page(t, w, fmt.Sprintf("slug-%02d", i), "own page")
		}
		s, e := webui.NewServer(w, nil)
		need(t, e)
		handler := s.Handler()
		start := make(chan struct{})
		codes := make(chan int, n)
		var wg sync.WaitGroup
		for i := 0; i < n; i++ {
			wg.Add(1)
			go func(i int) {
				defer wg.Done()
				<-start
				slug := fmt.Sprintf("slug-%02d", i)
				path := "/read/" + slug
				form := url.Values{}
				if door == "feedback" {
					path = "/resurface/feedback"
					form = url.Values{"slug": {slug}, "action": {"up"}}
				}
				request := httptest.NewRequest("POST", path, strings.NewReader(form.Encode()))
				request.Header.Set("Content-Type", "application/x-www-form-urlencoded")
				rw := httptest.NewRecorder()
				handler.ServeHTTP(rw, request)
				codes <- rw.Code
			}(i)
		}
		close(start)
		wg.Wait()
		close(codes)
		success, fail := 0, 0
		for code := range codes {
			if code == 303 {
				success++
			} else {
				fail++
			}
		}
		if door == "read" {
			st, e := reads.Load(w)
			need(t, e)
			t.Logf("same-server read mutex control: requests=%d success=%d failed=%d retained=%d", n, success, fail, len(st.Reads))
		} else {
			st, e := resurface.LoadState(w)
			retained := -1
			if e == nil {
				retained = len(st.Feedback)
			}
			t.Logf("actual feedback handler: requests=%d success=%d failed=%d retained=%d Load_error=%v", n, success, fail, retained, e)
		}
	}
}
