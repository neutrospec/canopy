from pathlib import Path
import os, subprocess, json, sys, time
root=Path(__file__).parent
env={k:v for k,v in os.environ.items() if k in ('PATH','LANG','LC_ALL','TZ','TERM')}
env.update(HOME=str(root/'home'),XDG_CONFIG_HOME=str(root/'config'),XDG_CACHE_HOME=str(root/'cache'),XDG_DATA_HOME=str(root/'data'),XDG_STATE_HOME=str(root/'state'),TMPDIR=str(root/'tmp'),GOCACHE=str(root/'go-cache'),GOMODCACHE='/Users/nobocop/go/pkg/mod',GOPROXY='off',GOSUMDB='off',GOTOOLCHAIN='local',GOTELEMETRY='off',CGO_ENABLED='0',GIT_CONFIG_NOSYSTEM='1',GIT_CONFIG_GLOBAL='/dev/null',PI_CODING_AGENT_DIR='',CANOPY_WIKI='')
cwd='/Users/nobocop/play/canopy'
def run(name,argv):
    start=time.time()
    cp=subprocess.run(argv,cwd=cwd,env=env,text=True,capture_output=True)
    rec=dict(name=name,argv=argv,exit=cp.returncode,seconds=round(time.time()-start,3),stdout=cp.stdout,stderr=cp.stderr)
    (root/'evidence'/f'{name}.json').write_text(json.dumps(rec,ensure_ascii=False,indent=2)+'\n')
    print(json.dumps(dict(name=name,exit=cp.returncode,seconds=rec['seconds'])),flush=True)
    print(cp.stdout[-7000:],flush=True)
    print(cp.stderr[-5000:],flush=True)
    return cp
