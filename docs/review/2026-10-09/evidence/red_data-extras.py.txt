from pathlib import Path
import os, subprocess, json, concurrent.futures, sqlite3, hashlib
ROOT=Path(os.environ['RED_DATA_ROOT']);BIN=Path(os.environ['RED_DATA_BIN']);ENV=dict(os.environ)
TRACE=Path('docs/review/2026-10-09/evidence/red_data-extra-transcript.jsonl').open('w')
def run(argv,env=None):
    argv=list(map(str,argv));r=subprocess.run(argv,env=env or ENV,text=True,capture_output=True,timeout=35)
    TRACE.write(json.dumps(dict(argv=argv,exit=r.returncode,stdout=r.stdout,stderr=r.stderr))+'\n');TRACE.flush();return r
def cli(w,*a):return run([BIN,'--wiki',w,*a])
def good(w,*a):
    r=cli(w,*a);assert not r.returncode,(a,r.stderr);return r
def git(w,*a):
    r=run(['git','-C',w,*a]);assert not r.returncode,(a,r.stderr);return r.stdout.strip()
def page(w,s,text):
    p=w/'concepts'/(s+'.md');p.parent.mkdir(parents=True,exist_ok=True);p.write_text(f'---\ntitle: {s}\ncreated: 2026-03-04\nupdated: 2026-03-04\ntype: concept\ntags: []\n---\n\n{text}\n')
