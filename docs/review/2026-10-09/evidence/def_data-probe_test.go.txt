func must(t *testing.T, e error) {
	t.Helper()
	if e != nil {
		t.Fatal(e)
	}
}
func write(t *testing.T, p, s string) {
	t.Helper()
	must(t, os.MkdirAll(filepath.Dir(p), 0755))
	must(t, os.WriteFile(p, []byte(s), 0644))
}
func read(t *testing.T, p string) []byte { t.Helper(); b, e := os.ReadFile(p); must(t, e); return b }
func newWiki(t *testing.T) *config.Wiki {
	t.Helper()
	p, e := os.MkdirTemp(filepath.Join(os.Getenv("DEF_DATA_ROOT"), "fixtures"), "go-")
	must(t, e)
	return &config.Wiki{Root: p, Cfg: config.Default()}
}
func pg(t *testing.T, w *config.Wiki, slug, body string) {
	t.Helper()
	write(t, filepath.Join(w.Root, "concepts", slug+".md"), wiki.NewPageContent(slug, "concept", "2026-02-01", []string{"tool"}, nil, body))
}
func scan(t *testing.T, w *config.Wiki) *wiki.ScanResult {
	t.Helper()
	s, e := wiki.Scan(w)
	must(t, e)
	return s
}
func rd(t *testing.T, w *config.Wiki) *reads.State {
	t.Helper()

	must(t, e)
	return s
}
func surf(t *testing.T, w *config.Wiki) *resurface.State {
	t.Helper()

	must(t, e)

}
func post(h http.Handler, path string, v url.Values) int {
	r := httptest.NewRequest("POST", path, strings.NewReader(v.Encode()))
	r.Header.Set("Content-Type", "application/x-www-form-urlencoded")
	rec := httptest.NewRecorder()
	h.ServeHTTP(rec, r)
	return rec.Code
}

	log.SetOutput(io.Discard)
	w := newWiki(t)
	pg(t, w, "alpha", "original body")
	before := reconcile.HashBytes(read(t, filepath.Join(w.Root, "concepts/alpha.md")))
	s, e := webui.NewServer(w, nil)
	must(t, e)
	h := s.Handler()
	for i := 0; i < 5; i++ {
		if post(h, "/edit/alpha", url.Values{"body": {fmt.Sprintf("sequential-%02d", i)}}) != 303 {
			t.Fatal("control")
		}

	must(t, e)

	codes := make(chan int, n)
	var wg sync.WaitGroup
	for i := 0; i < n; i++ {
		wg.Add(1)
		go func(i int) {
			defer wg.Done()
			<-start
			codes <- post(h, "/edit/alpha", url.Values{"body": {fmt.Sprintf("different-body-%03d", i)}})
		}(i)
	}
	close(start)
	wg.Wait()
	close(codes)

		if c == 303 {
			success++
		}
	}

	must(t, e)

			bad++
			continue
		}
		if strings.HasPrefix(x.Body, "different-body-") {
			bodies[x.Body] = true
		}
	}
	t.Logf("G01 requests=%d HTTP303=%d retained_distinct_bodies=%d corrupt=%d page_unchanged=%v", n, success, len(bodies), bad, before == reconcile.HashBytes(read(t, filepath.Join(w.Root, "concepts/alpha.md"))))

	must(t, e)
	t.Logf("G01 independent_same_time_equal_ID=%v (API only; no clone merge)", x.ID == y.ID)
}

func TestG02HTTPStateAndMutexControls(t *testing.T) {
	for _, door := range []string{"read-one-server", "read-two-servers", "feedback"} {
		w := newWiki(t)
		const n = 28
		for i := 0; i < n; i++ {
			pg(t, w, fmt.Sprintf("p-%02d", i), "body")
		}
		must(t, surf(t, w).Save(w))
		s, e := webui.NewServer(w, nil)
		must(t, e)
		h := s.Handler()
		s2, e := webui.NewServer(w, nil)
		must(t, e)
		h2 := s2.Handler()
		start := make(chan struct{})
		var wg sync.WaitGroup
		success := make(chan string, n)
		codes := make(chan int, n)
		for i := 0; i < n; i++ {
			wg.Add(1)
			go func(i int) {
				defer wg.Done()
				<-start
				slug := fmt.Sprintf("p-%02d", i)
				handler := h
				path := "/read/" + slug
				v := url.Values{}
				if door == "read-two-servers" && i%2 == 1 {
					handler = h2
				}
				if door == "feedback" {
					path = "/resurface/feedback"
					v = url.Values{"slug": {slug}, "action": {"up"}}
				}
				c := post(handler, path, v)
				codes <- c
				if c == 303 {
					success <- slug
				}
			}(i)
		}
		close(start)
		wg.Wait()
		close(success)
		close(codes)
		successSet := map[string]bool{}
		status := map[int]int{}
		for c := range codes {
			status[c]++
		}
		for k := range success {
			successSet[k] = true
		}
		retained := map[string]bool{}
		var loadErr error
		if door == "feedback" {
			st, e := resurface.LoadState(w)
			loadErr = e
			if e == nil {
				for _, f := range st.Feedback {
					retained[f.Slug] = true
				}
			}
		} else {
			st, e := reads.Load(w)
			loadErr = e
			if e == nil {
				for k := range st.Reads {
					retained[k] = true
				}
			}
		}
		lost := 0
		for k := range successSet {
			if !retained[k] {
				lost++
			}
		}
		t.Logf("G02 %s statuses=%v successful_distinct=%d retained=%d successful_missing=%d Load_error=%v", door, status, len(successSet), len(retained), lost, loadErr)
	}
}

func TestG05MigrationShapeAndConcurrency(t *testing.T) {
	for _, input := range []string{"missing", "null", `{"schema_version":-3}`, `{"schema_version":88}`, "{"} {
		w := newWiki(t)
		ctx := &migrate.Context{ConfigHome: filepath.Join(w.Root, "config"), DataHome: filepath.Join(w.Root, "data"), CacheHome: filepath.Join(w.Root, "cache")}
		path := filepath.Join(ctx.ConfigHome, "state.json")
		if input != "missing" {
			write(t, path, input)
		}
		r, e := migrate.Ensure(ctx, "def-audit")
		changed := false
		if b, re := os.ReadFile(path); re == nil {
			changed = string(b) != input
		}
		r2, e2 := migrate.Ensure(ctx, "def-audit")
		t.Logf("G05 shape=%s result=%+v err=%v changed=%v second=%+v second_err=%v", input, r, e, changed, r2, e2)
	}
	w := newWiki(t)
	ctx := &migrate.Context{ConfigHome: filepath.Join(w.Root, "config"), DataHome: filepath.Join(w.Root, "data"), CacheHome: filepath.Join(w.Root, "cache")}
	write(t, filepath.Join(ctx.ConfigHome, "state.json"), `{"schema_version":0}`)
	const n = 32
	start := make(chan struct{})
	errs := make(chan error, n)
	var wg sync.WaitGroup
	for i := 0; i < n; i++ {
		wg.Add(1)
		go func() { defer wg.Done(); <-start; _, e := migrate.Ensure(ctx, "def-audit"); errs <- e }()
	}
	close(start)
	wg.Wait()
	close(errs)
	failed := 0
	sample := ""
	for e := range errs {
		if e != nil {
			failed++
			sample = e.Error()
		}
	}
	t.Logf("G05 goroutine_Ensure calls=%d failed=%d sample=%s", n, failed, sample)
}

	t.Logf("G11 alias_event_key_same=%v checkout_key_same=%v alias_open=%d original_copy_survives=%v", w.AttentionDBPath() == other.AttentionDBPath(), w.CheckoutDir() == other.CheckoutDir(), len(opens), ce == nil)
	path := filepath.Join(w.Root, "future.db")
	ev, e := attention.Open(path)
	must(t, e)
	must(t, ev.Log(fixed, "alpha", "agent", "show", "prior"))
	must(t, ev.Close())
	raw, e := sql.Open("sqlite", path)
	must(t, e)
	_, e = raw.Exec("PRAGMA user_version = 8")
	must(t, e)
	must(t, raw.Close())
	ev, e = attention.Open(path)
	must(t, e)
	le := ev.Log(fixed, "beta", "agent", "show", "later")
	rows, e := ev.Recent(9)
	must(t, e)
	must(t, ev.Close())
	t.Logf("G11 higher_version_same_real_schema Log=%v rows=%d (no actual future schema)", le, len(rows))
}

func TestG12CLIHashWriteWindow(t *testing.T) {
	for _, concurrent := range []bool{false, true} {
		w := newWiki(t)
		pg(t, w, "alpha", "baseline")
		path := filepath.Join(w.Root, "concepts/alpha.md")
		co := exec.Command(os.Getenv("DEF_DATA_BIN"), "--wiki", w.Root, "checkout", "alpha", "--json")
		out, e := co.CombinedOutput()
		must(t, e)
		var info checkout.Info
		must(t, json.Unmarshal(out, &info))
		put := string(read(t, info.Path)) + strings.Repeat("\n```mermaid\nflowchart LR\n    AA[\"source\"] --> BB[\"target\"]\n```\n", 24)
		if len(mermaid.Blocks(put)) != 24 {
			t.Fatal("slow validation fixture has wrong block count")
		}
		write(t, info.Path, put)
		sentinel := "\nDIRECT-WRITER-SENTINEL\n"
		if !concurrent {
			write(t, path, string(read(t, path))+sentinel)
		}
		cmd := exec.Command(os.Getenv("DEF_DATA_BIN"), "--wiki", w.Root, "checkin", "alpha", "--json")
		start := time.Now()
		done := make(chan struct {
			out []byte
			e   error
		}, 1)
		go func() {
			o, e := cmd.CombinedOutput()
			done <- struct {
				out []byte
				e   error
			}{o, e}
		}()
		if concurrent {
			time.Sleep(150 * time.Millisecond)
			write(t, path, string(read(t, path))+sentinel)
		}
		r := <-done
		_, copyErr := os.Stat(info.Path)
		t.Logf("G12 actual_CLI concurrent=%v elapsed=%v err=%v direct_writer_preserved=%v copy_present=%v output=%s", concurrent, time.Since(start), r.e, strings.Contains(string(read(t, path)), "DIRECT-WRITER-SENTINEL"), copyErr == nil, r.out)
	}
}
