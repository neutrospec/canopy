def event(kind,**values):
 data=json.loads(LOG.read_text()) if LOG.exists() else {'created_utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),'environment':ENV,'root':str(ROOT),'events':[]}
 item={'id':len(data['events'])+1,'kind':kind,'utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),**values}

def run(argv,cwd=REPO,input=None,required=False):
 argv=[str(x) for x in argv]
 p=subprocess.run(argv,cwd=cwd,env=ENV,input=input,text=True,stdout=subprocess.PIPE,stderr=subprocess.PIPE)
 event('command',argv=argv,cwd=str(cwd),stdin=input,stdout=p.stdout,stderr=p.stderr,exit_code=p.returncode)
 if required and p.returncode: raise RuntimeError('command failed: '+repr(argv))
 return p
def canopy(*args,required=False): return run([BIN,*args],required=required)
def decide(text): event('decision',text=text)
def observe(path):
 path=pathlib.Path(path)
 if not path.exists(): return event('observed_file',path=str(path),exists=False)
 b=path.read_bytes()
 return event('observed_file',path=str(path),exists=True,sha256=hashlib.sha256(b).hexdigest(),content=b.decode('utf-8'),mode=oct(path.stat().st_mode & 0o777))
def write(path,content,mode=None):
 path=pathlib.Path(path);path.parent.mkdir(parents=True,exist_ok=True);path.write_text(content)
 if mode is not None: path.chmod(mode)
 event('file_write',path=str(path),content=content,sha256=hashlib.sha256(path.read_bytes()).hexdigest(),mode=oct(path.stat().st_mode & 0o777))
def snapshot_tree(path,label):
 path=pathlib.Path(path)
 files={str(p.relative_to(path)):hashlib.sha256(p.read_bytes()).hexdigest() for p in sorted(path.rglob('*')) if p.is_file() and '.git' not in p.relative_to(path).parts}
 return event('tree_hashes',label=label,path=str(path),files=files)
